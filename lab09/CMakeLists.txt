cmake_minimum_required(VERSION 3.20)
project(SeguimientoAcademico LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(seguimiento main.cpp MainWindow.h)
target_link_libraries(seguimiento PRIVATE Qt6::Widgets)

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(pruebas tests/pruebas.cpp)
    target_include_directories(pruebas PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(pruebas PRIVATE Qt6::Test)
    add_test(NAME modelo_controlador COMMAND pruebas)
    add_test(NAME integracion_gui COMMAND seguimiento --smoke
        ${CMAKE_CURRENT_SOURCE_DIR}/datos/actividades.csv)
    set_tests_properties(integracion_gui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
